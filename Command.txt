1. DYNAMODB TABLE CREATION & PROVISIONING
-------------------------------------------------------------------
awslocal dynamodb create-table \
  --table-name coupons \
  --attribute-definitions \
      AttributeName=id,AttributeType=S \
      AttributeName=provider_id,AttributeType=S \
      AttributeName=campaign_id,AttributeType=S \
  --key-schema \
      AttributeName=id,KeyType=HASH \
  --provisioned-throughput ReadCapacityUnits=25,WriteCapacityUnits=10 \
  --global-secondary-indexes \
      "[
        {
          \"IndexName\": \"provider-campaign-index\",
          \"KeySchema\": [
            {\"AttributeName\":\"provider_id\",\"KeyType\":\"HASH\"},
            {\"AttributeName\":\"campaign_id\",\"KeyType\":\"RANGE\"}
          ],
          \"Projection\": {\"ProjectionType\":\"ALL\"},
          \"ProvisionedThroughput\": {\"ReadCapacityUnits\": 5, \"WriteCapacityUnits\": 5}
        }
      ]"
